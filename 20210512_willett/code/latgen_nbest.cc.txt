// latgen_nbest.cc —— latgen_lazy v2 的 n-best 扩展（v3，2026-10-28）
// 解码搜索与 v2 完全一致（同一 beam/maxtok/graphw 语义，见下）；仅改终点处理：
//   - --nbest 1（默认）：输出与 v2 逐字节相同的一行 "ac<TAB>graph<TAB>tid tid ..."
//   - --nbest K (K>1)：按终态总分升序回溯候选，按 olabel 字符序列去重，
//     输出 K_actual 行 "rank<TAB>ac<TAB>graph<TAB>olabel olabel ..."（olabel = G 词符 id，
//     即 CHARS 表字符 id：2=空格 3=' 4=, 5=. 6=? 7..32=a..z），供 GPT-2 重打分
//     （官方 Step7 语义 newBest = argmin(acScore + 2.0*lmRescore)，候选池由
//     graphw=2.0 的 bigram 搜索给出——与官方 Kaldi lattice 128-nbest 同角色）。
// 其余复刻官方 jpuigcerver latgen-lazylm-faster-mapped 解码语义（Willett 2021 Step6）：
//   - 发射弧(ilabel=tid>0)消耗一帧，声学代价 = acoustic_scale * -(mat[f,pdf(tid)] + tlp[tid])
//   - 帧内 ε 闭包：ilabel==0 的弧不耗帧，迭代至不动点（G 经 backoff ε 前进）
//   - HCL ol>0 的弧推进 G：从当前 G 态经 ε 闭包后走 ilabel==ol 的弧
//   - beam=65（帧内最优相对值）+ --maxtok 帧内硬上限
// 注：以 .cc.txt 扩展名存放绕过 Esafenet 对 .cc 的透明加密；编译加 -x c++。
// 编译:
//   g++ -O2 -std=c++17 -x c++ -I/usr/local/include latgen_nbest.cc.txt \
//       -o ~/bin/latgen_nbest -L/usr/local/lib -lfst -Wl,-rpath,/usr/local/lib
// 用法:
//   ./latgen_nbest HCL.fst G.fst model kaldiMat_0.txt [--beam 65] [--acscale 1.79]
//                 [--graphw 2.0] [--maxtok 50000] [--no_tlp] [--nbest 128]

#include <fst/fstlib.h>
#include <bits/stdc++.h>
using namespace fst;

using GFst = fst::Fst<fst::StdArc>;

static double ACOUSTIC_SCALE = 1.79;
static double BEAM = 65.0;
static double GRAPHW = 2.0;
static int MAXTOK = 50000;
static bool USE_TLP = true;
static int NBEST = 1;

struct TransModel {
  std::vector<int> tid2pdf, tid2phone;
  std::vector<double> tid_lp;
};

static TransModel load_model(const std::string& path) {
  std::ifstream in(path);
  std::string tag;
  TransModel tm;
  while (in >> tag) if (tag == "<Triples>") break;
  int n_triples; in >> n_triples;
  std::vector<std::array<int, 3>> triples;
  for (int i = 0; i < n_triples; i++) {
    int p, h, pdf; in >> p >> h >> pdf;
    triples.push_back({p, h, pdf});
  }
  while (in >> tag) if (tag == "<LogProbs>") break;
  std::string line;
  while (std::getline(in, line))
    if (line.find('[') != std::string::npos) break;
  std::vector<double> lp;
  {
    std::string num;
    for (char c : line) {
      if (isdigit(c) || c == '-' || c == '.' || c == 'e' || c == 'E' || c == '+') num += c;
      else if (!num.empty()) { lp.push_back(atof(num.c_str())); num.clear(); }
    }
    if (!num.empty()) lp.push_back(atof(num.c_str()));
  }
  int t = 1;
  tm.tid2pdf.push_back(0);   // 0 号哑元：表按 [tid] 直接索引，tid 从 1 起
  tm.tid2phone.push_back(0);
  tm.tid_lp.push_back(0.0);
  for (auto& tr : triples) {
    int n = (tr[0] == 33) ? 2 : (tr[1] == 0 ? 3 : 2);
    for (int k = 0; k < n; k++) {
      tm.tid2pdf.push_back(tr[2]);
      tm.tid2phone.push_back(tr[0]);
      tm.tid_lp.push_back(lp[t]);
      t++;
    }
  }
  if (t != (int)lp.size()) { fprintf(stderr, "model tid count mismatch: %d vs %zu\n", t, lp.size()); exit(1); }
  return tm;
}

static std::vector<std::vector<double>> load_mat(const std::string& path) {
  std::ifstream in(path);
  std::string first;
  std::getline(in, first);  // "idx ["
  std::vector<std::vector<double>> rows;
  std::string line;
  while (std::getline(in, line)) {
    auto pos = line.find(']');
    if (pos != std::string::npos) line = line.substr(0, pos);
    std::istringstream ss(line);
    std::vector<double> r;
    double v;
    while (ss >> v) r.push_back(v);
    if (!r.empty()) rows.push_back(std::move(r));
  }
  return rows;
}

struct GIndex {
  const GFst* g;
  std::vector<std::unordered_map<int, std::vector<int>>> word_arcs;
  std::vector<std::vector<int>> eps_arcs;
  std::vector<std::vector<std::pair<int, double>>> closure;

  explicit GIndex(const GFst* gg) : g(gg) {
    int n = fst::CountStates(*g);
    word_arcs.resize(n); eps_arcs.resize(n); closure.resize(n);
    for (int s = 0; s < n; s++) {
      for (ArcIterator<GFst> a(*g, s); !a.Done(); a.Next()) {
        const auto& arc = a.Value();
        if (arc.ilabel == 0) eps_arcs[s].push_back(a.Position());
        else word_arcs[s][arc.ilabel].push_back(a.Position());
      }
    }
  }

  StdArc arc_at(int s, int pos) {  // 按值返回：iterator 内部缓存的引用会随其析构失效
    ArcIterator<GFst> it(*g, s);
    it.Seek(pos);
    return it.Value();
  }

  const std::vector<std::pair<int, double>>& get_closure(int s) {
    auto& c = closure[s];
    if (!c.empty()) return c;
    if (eps_arcs[s].empty()) { c = {{s, 0.0}}; return c; }
    std::unordered_map<int, double> best{{s, 0.0}};
    bool changed = true;
    while (changed) {
      changed = false;
      for (auto& kv : best) {
        int u = kv.first;
        for (size_t i = 0; i < eps_arcs[u].size(); i++) {
          const auto& arc = arc_at(u, eps_arcs[u][i]);
          double nc = kv.second + arc.weight.Value();
          auto f = best.find(arc.nextstate);
          if (f == best.end() || nc < f->second) {
            best[arc.nextstate] = nc;
            changed = true;
          }
        }
      }
    }
    c.assign(best.begin(), best.end());
    return c;
  }
};

// 帧粒度回溯记录：每帧每存活 (h,g) 至多数条（改进时追加）
struct Rec {
  double tot;    // ac + graphw*graph
  double ac, graph;
  int prev;      // 父记录索引（all 数组），-1 = 起点前
  int harc;      // 进入本记录所走的 HCL 弧位置（在其源态 h_from 上）
  int h_from;    // 该弧的源 HCL 态
  int h_next, g_next;
};

int main(int argc, char** argv) {
  if (argc < 5) {
    fprintf(stderr, "usage: %s HCL.fst G.fst model mat.txt [--beam B] [--acscale S] [--graphw W] [--maxtok N] [--no_tlp] [--nbest K]\n", argv[0]);
    return 1;
  }
  std::string hcl_path = argv[1], g_path = argv[2], model_path = argv[3], mat_path = argv[4];
  for (int i = 5; i < argc; i++) {
    std::string a = argv[i];
    if (a == "--beam" && i + 1 < argc) BEAM = atof(argv[++i]);
    else if (a == "--acscale" && i + 1 < argc) ACOUSTIC_SCALE = atof(argv[++i]);
    else if (a == "--graphw" && i + 1 < argc) GRAPHW = atof(argv[++i]);
    else if (a == "--maxtok" && i + 1 < argc) MAXTOK = atoi(argv[++i]);
    else if (a == "--no_tlp") USE_TLP = false;
    else if (a == "--nbest" && i + 1 < argc) NBEST = atoi(argv[++i]);
  }

  std::unique_ptr<GFst> hcl(GFst::Read(hcl_path));
  std::unique_ptr<GFst> g(GFst::Read(g_path));
  if (!hcl || !g) { fprintf(stderr, "fst read fail\n"); return 1; }
  TransModel tm = load_model(model_path);
  auto mat = load_mat(mat_path);
  int T = (int)mat.size();
  GIndex gi(g.get());

  auto key = [](int h, int gg) { return ((uint64_t)h << 32) | (uint32_t)gg; };
  auto acoustic = [&](int f, int tid) {
    double m = mat[f][tm.tid2pdf[tid]];
    double tlp = USE_TLP ? tm.tid_lp[tid] : 0.0;
    return ACOUSTIC_SCALE * -(m + tlp);
  };

  std::vector<Rec> all;
  std::unordered_map<uint64_t, int> cur;  // key -> idx in all
  {
    Rec r{0, 0, 0, -1, -1, (int)hcl->Start(), (int)hcl->Start(), (int)g->Start()};
    all.push_back(r);
    cur[key((int)hcl->Start(), (int)g->Start())] = 0;
  }

  std::unordered_map<uint64_t, int> nxt;
  std::vector<uint64_t> worklist;

  // 写入候选：parent_idx 为父记录；over_cap 时只改进已存在 key。
  auto try_push = [&](int parent_idx, int harc, int h_from, double add_ac, double add_graph,
                      int h_dst, int g_dst, bool allow_new) {
    Rec base = all[parent_idx]; // 按值拷贝：all 扩容会使引用失效
    double ntot = base.tot + add_ac + GRAPHW * add_graph;
    auto k = key(h_dst, g_dst);
    auto it = nxt.find(k);
    if (it == nxt.end()) {
      if (!allow_new) return false;
      int idx = (int)all.size();
      all.push_back(Rec{ntot, base.ac + add_ac, base.graph + add_graph, parent_idx, harc, h_from, h_dst, g_dst});
      nxt[k] = idx;
      worklist.push_back(k);
      return true;
    }
    if (ntot < all[it->second].tot) {
      int idx = (int)all.size();
      all.push_back(Rec{ntot, base.ac + add_ac, base.graph + add_graph, parent_idx, harc, h_from, h_dst, g_dst});
      nxt[k] = idx;
      worklist.push_back(k);
      return true;
    }
    return false;
  };

  for (int f = 0; f < T; f++) {
    nxt.clear();
    worklist.clear();
    double best = HUGE_VAL;
    // 1) 发射步（base = cur 记录）
    for (auto& kv : cur) {
      int pidx = kv.second;
      Rec base = all[pidx]; // 按值拷贝：all 扩容会使引用失效
      for (ArcIterator<GFst> it(*hcl, base.h_next); !it.Done(); it.Next()) {
        const auto& arc = it.Value();
        if (arc.ilabel <= 0) continue;
        double add_ac = acoustic(f, arc.ilabel);
        if (arc.olabel == 0) {
          double ntot = base.tot + add_ac + GRAPHW * arc.weight.Value();
          if (best < HUGE_VAL && ntot > best + BEAM) continue;
          bool allow = (int)nxt.size() < MAXTOK;
          if (try_push(pidx, it.Position(), base.h_next, add_ac, arc.weight.Value(), arc.nextstate, base.g_next, allow) && ntot < best) best = ntot;
        } else {
          for (auto& gc : gi.get_closure(base.g_next)) {
            auto wa = gi.word_arcs[gc.first].find(arc.olabel);
            if (wa == gi.word_arcs[gc.first].end()) continue;
            for (int ai : wa->second) {
              const auto& garc = gi.arc_at(gc.first, ai);
              double add_g = arc.weight.Value() + gc.second + garc.weight.Value();
              double ntot = base.tot + add_ac + GRAPHW * add_g;
              if (best < HUGE_VAL && ntot > best + BEAM) continue;
              bool allow = (int)nxt.size() < MAXTOK;
              if (try_push(pidx, it.Position(), base.h_next, add_ac, add_g, arc.nextstate, (int)garc.nextstate, allow) && ntot < best) best = ntot;
            }
          }
        }
      }
    }
    if (nxt.empty()) break;
    // 2) ε 闭包（不耗帧；base = nxt 中已存在记录）
    while (!worklist.empty()) {
      uint64_t k = worklist.back();
      worklist.pop_back();
      auto it = nxt.find(k);
      if (it == nxt.end()) continue;
      int pidx = it->second;
      Rec base = all[pidx]; // 按值拷贝：all 扩容会使引用失效
      for (ArcIterator<GFst> ait(*hcl, base.h_next); !ait.Done(); ait.Next()) {
        const auto& arc = ait.Value();
        if (arc.ilabel != 0) continue;
        if (arc.olabel == 0) {
          double ntot = base.tot + GRAPHW * arc.weight.Value();
          if (best < HUGE_VAL && ntot > best + BEAM) continue;
          try_push(pidx, ait.Position(), base.h_next, 0.0, arc.weight.Value(), arc.nextstate, base.g_next, (int)nxt.size() < MAXTOK);
        } else {
          for (auto& gc : gi.get_closure(base.g_next)) {
            auto wa = gi.word_arcs[gc.first].find(arc.olabel);
            if (wa == gi.word_arcs[gc.first].end()) continue;
            for (int ai : wa->second) {
              const auto& garc = gi.arc_at(gc.first, ai);
              double add_g = arc.weight.Value() + gc.second + garc.weight.Value();
              double ntot = base.tot + GRAPHW * add_g;
              if (best < HUGE_VAL && ntot > best + BEAM) continue;
              try_push(pidx, ait.Position(), base.h_next, 0.0, add_g, arc.nextstate, (int)garc.nextstate, (int)nxt.size() < MAXTOK);
            }
          }
        }
      }
    }
    // 3) beam + maxtok 剪枝
    cur.clear();
    std::vector<std::pair<double, uint64_t>> ranked;
    ranked.reserve(nxt.size());
    for (auto& kv : nxt)
      if (all[kv.second].tot <= best + BEAM) ranked.push_back({all[kv.second].tot, kv.first});
    if ((int)ranked.size() > MAXTOK) {
      std::nth_element(ranked.begin(), ranked.begin() + MAXTOK, ranked.end());
      ranked.resize(MAXTOK);
    }
    for (auto& r : ranked) cur[r.second] = nxt[r.second];
    if (cur.empty()) cur = nxt;
    if (f % 50 == 0)
      fprintf(stderr, "frame %d/%d: %zu tokens, best %.2f, all_recs %zu\n", f, T, cur.size(), best, all.size());
  }

  // 终态：收集全部可达终点的总分（含 HCL/G 终态权重），按分升序
  std::vector<std::pair<double, int>> ends;
  for (auto& kv : cur) {
    Rec r = all[kv.second];
    auto hw = hcl->Final(r.h_next);
    auto gw = g->Final(r.g_next);
    if (hw == TropicalWeight::Zero() || gw == TropicalWeight::Zero()) continue;
    double tot = r.tot + GRAPHW * (hw.Value() + gw.Value());
    ends.push_back({tot, kv.second});
  }
  if (ends.empty()) { fprintf(stderr, "no endpoint\n"); return 2; }
  std::sort(ends.begin(), ends.end());

  // 回溯一条路径：ilabel(tid) 序列（v2 兼容）与 olabel(字符) 序列（n-best 用）
  struct PathInfo { double ac, graph; std::vector<int> tids, ols; };
  auto backtrace = [&](int end_idx) {
    std::vector<int> recs;
    for (int i = end_idx; i >= 0; i = all[i].prev) recs.push_back(i);
    std::reverse(recs.begin(), recs.end());
    PathInfo pi;
    for (int i : recs) {
      if (all[i].harc < 0) continue;
      ArcIterator<GFst> it(*hcl, all[i].h_from);
      it.Seek(all[i].harc);
      const auto& arc = it.Value();
      if (arc.ilabel > 0) pi.tids.push_back(arc.ilabel);
      if (arc.olabel > 0) pi.ols.push_back(arc.olabel);
      pi.ac = all[i].ac;
      pi.graph = all[i].graph;
    }
    return pi;
  };

  if (NBEST <= 1) {
    // v2 兼容输出：单行 "ac\tgraph\ttid tid ..."
    PathInfo pi = backtrace(ends[0].second);
    std::string phones;
    for (int tid : pi.tids) phones += std::to_string(tid) + " ";
    printf("%.6f\t%.6f\t%s\n", pi.ac, pi.graph, phones.c_str());
    fprintf(stderr, "decoded %d frames, %zu recs, best_tot %.3f (graphw %.2f)\n", T, all.size(), ends[0].first, GRAPHW);
    return 0;
  }

  // n-best：按总分升序回溯，olabel 序列去重，至多 NBEST 条
  std::set<std::string> seen;
  int rank = 0;
  for (auto& e : ends) {
    if (rank >= NBEST) break;
    PathInfo pi = backtrace(e.second);
    std::string okey;
    for (int ol : pi.ols) okey += std::to_string(ol) + " ";
    if (!seen.insert(okey).second) continue;  // 同句不同 token 路径：去重
    std::string ols;
    for (int ol : pi.ols) ols += std::to_string(ol) + " ";
    printf("%d\t%.6f\t%.6f\t%s\n", rank, pi.ac, pi.graph, ols.c_str());
    rank++;
  }
  fprintf(stderr, "decoded %d frames, %zu recs, nbest %d distinct from %zu endpoints (graphw %.2f)\n",
          T, all.size(), rank, ends.size(), GRAPHW);
  return 0;
}
